무인 서버측 자동 백업 (Supabase)

이 화면을 닫아도 동작하는 완전 무인 백업은 Supabase의 예약 작업으로 구성합니다. pg_cron + Edge Function으로 매일 전일 데이터를 집계해 env_backup_log 테이블에 적재하도록 예약하면, PC가 꺼져 있어도 서버에서 자동 실행됩니다. 아래 SQL을 Supabase SQL Editor에서 한 번 실행해 테이블을 준비하세요.

-- 백업 기록 테이블 (1회 실행)
create table if not exists env_backup_log (
  id bigint generated always as identity primary key,
  backup_date date not null,
  row_count int not null default 0,
  over_count int not null default 0,
  t_avg numeric, h_avg numeric,
  format text default 'xlsx',
  source text default 'client',
  operator text,
  created_at timestamptz default now()
);
 
-- (선택) 익명 INSERT 허용 RLS 정책
alter table env_backup_log enable row level security;
create policy "insert_anon" on env_backup_log
  for insert to anon with check (true);
create policy "read_anon" on env_backup_log
  for select to anon using (true);
