▶ 📋 Supabase SQL 스키마 (최초 1회 실행) 
-- ===== 1. 측정점 마스터 =====
CREATE TABLE IF NOT EXISTS illuminance_point (
  id            bigint generated always as identity primary key,
  code          text unique not null,
  area          text not null,
  point_name    text not null,
  standard_lux  numeric not null,
  sort_order    int default 0,
  is_active     boolean default true,
  note          text,
  created_at    timestamptz default now(),
  updated_at    timestamptz default now()
);
CREATE INDEX IF NOT EXISTS idx_illum_point_active_sort
  ON illuminance_point(is_active, sort_order);
ALTER TABLE illuminance_point ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "anon_all" ON illuminance_point;
CREATE POLICY "anon_all" ON illuminance_point FOR ALL TO anon USING (true) WITH CHECK (true);

-- ===== 2. 측정 기록 =====
CREATE TABLE IF NOT EXISTS illuminance_check (
  id            bigint generated always as identity primary key,
  check_date    date not null,
  check_time    time not null,
  inspector     text not null,
  area          text not null,
  point_code    text not null,
  point_name    text not null,
  lux_value     numeric not null,
  standard_lux  numeric not null,
  result        text not null,
  ng_count      int default 0,
  memo          text,
  created_at    timestamptz default now()
);
CREATE INDEX IF NOT EXISTS idx_illum_check_date
  ON illuminance_check(check_date DESC);
CREATE INDEX IF NOT EXISTS idx_illum_check_area_date
  ON illuminance_check(area, check_date DESC);
ALTER TABLE illuminance_check ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "anon_all" ON illuminance_check;
CREATE POLICY "anon_all" ON illuminance_check FOR ALL TO anon USING (true) WITH CHECK (true);
